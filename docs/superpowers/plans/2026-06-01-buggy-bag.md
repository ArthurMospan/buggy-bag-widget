# BuggyBag NPM Package Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Build a standalone React NPM package (`buggy-bag`) that renders a ghost QA overlay — screenshot + draw annotations + bug inbox + AI prompt generation — only when `localStorage.getItem('BUGGY_BAG_ACCESS') === 'active'`.

**Architecture:** A single `<BuggyBag />` component guards itself with a localStorage check, then renders a floating bug icon that opens a fullscreen capture/draw mode (html2canvas + react-konva) and a dashboard (Zustand-persisted bug list). All styles are self-contained via tsup's `injectStyle` + a standalone Tailwind v3 build that scans only the package's own src.

**Tech Stack:** React 18+, TypeScript, Tailwind CSS v3 (standalone in-package), tsup, Zustand (with persist), html2canvas, react-konva, lucide-react, Web Speech API

```ts
import { create } from 'zustand';
import { persist } from 'zustand/middleware';
import type { Bug, BugStatus } from './types';

interface BugStore {
  bugs: Bug[];
  addBug: (bug: Bug) => void;
  updateBugStatus: (id: string, status: BugStatus) => void;
  removeBug: (id: string) => void;
}

export const useBugStore = create<BugStore>()(
  persist(
    (set) => ({
      bugs: [],
      addBug: (bug) =>
        set((state) => ({ bugs: [bug, ...state.bugs] })),
      updateBugStatus: (id, status) =>
        set((state) => ({
          bugs: state.bugs.map((b) => (b.id === id ? { ...b, status } : b)),
        })),
      removeBug: (id) =>
        set((state) => ({ bugs: state.bugs.filter((b) => b.id !== id) })),
    }),
    {
      name: 'buggy-bag-storage',
      partialize: (state) => ({ bugs: state.bugs }),
    }
  )
);
```

```tsx
export function isGodModeActive(): boolean {
```

```tsx
import React, { useRef, useState, useCallback } from 'react';
import { Stage, Layer, Rect, Arrow, Circle, Text, Group } from 'react-konva';
import type Konva from 'konva';
import type { DrawShape, DrawTool } from '../types';

interface DrawingCanvasProps {
  width: number;
  height: number;
  tool: DrawTool;
  shapes: DrawShape[];
  onShapeComplete: (shape: DrawShape) => void;
}

let pinCounter = 1;

function uid(): string {
  return Math.random().toString(36).slice(2, 9);
}

function getStagePos(e: Konva.KonvaEventObject<MouseEvent>): { x: number; y: number } {
  const pos = e.target.getStage()?.getPointerPosition();
  return pos ?? { x: 0, y: 0 };
}

export function DrawingCanvas({
  width,
  height,
  tool,
  shapes,
  onShapeComplete,
}: DrawingCanvasProps) {
  const [draft, setDraft] = useState<DrawShape | null>(null);
  const isDrawing = useRef(false);
  const origin = useRef({ x: 0, y: 0 });

  const handleMouseDown = useCallback(
    (e: Konva.KonvaEventObject<MouseEvent>) => {
      const { x, y } = getStagePos(e);
      isDrawing.current = true;
      origin.current = { x, y };

      if (tool === 'pin') {
        onShapeComplete({ id: uid(), type: 'pin', x, y, pinNumber: pinCounter++ });
        isDrawing.current = false;
        return;
      }

      setDraft({
        id: uid(),
        type: tool,
        x,
        y,
        ...(tool === 'rect'
          ? { width: 0, height: 0 }
          : { points: [x, y, x, y] }),
      });
    },
    [tool, onShapeComplete]
  );

  const handleMouseMove = useCallback(
    (e: Konva.KonvaEventObject<MouseEvent>) => {
      if (!isDrawing.current || !draft) return;
      const { x, y } = getStagePos(e);

      if (tool === 'rect') {
        setDraft((d) =>
          d
            ? { ...d, width: x - origin.current.x, height: y - origin.current.y }
            : d
        );
      } else if (tool === 'arrow') {
        setDraft((d) =>
          d ? { ...d, points: [origin.current.x, origin.current.y, x, y] } : d
        );
      }
    },
    [draft, tool]
  );

  const handleMouseUp = useCallback(() => {
    if (!isDrawing.current || !draft) return;
    isDrawing.current = false;
    const tooSmall =
      tool === 'rect' &&
      Math.abs(draft.width ?? 0) < 10 &&
      Math.abs(draft.height ?? 0) < 10;
    if (!tooSmall) onShapeComplete(draft);
    setDraft(null);
  }, [draft, tool, onShapeComplete]);

  return (
    <Stage
      width={width}
      height={height}
      style={{ position: 'absolute', inset: 0, cursor: 'crosshair' }}
      onMouseDown={handleMouseDown}
      onMouseMove={handleMouseMove}
      onMouseUp={handleMouseUp}
    >
      <Layer>
        {shapes.map(renderShape)}
        {draft && renderShape(draft)}
      </Layer>
    </Stage>
  );
}

function renderShape(s: DrawShape) {
  if (s.type === 'rect') {
    return (
      <Rect
        key={s.id}
        x={s.x}
        y={s.y}
        width={s.width ?? 0}
        height={s.height ?? 0}
        stroke="#ef4444"
        strokeWidth={2}
        fill="rgba(239,68,68,0.08)"
        dash={[6, 3]}
      />
    );
  }
  if (s.type === 'arrow') {
    return (
      <Arrow
        key={s.id}
        points={s.points ?? []}
        stroke="#ef4444"
        strokeWidth={2.5}
        fill="#ef4444"
        pointerLength={10}
        pointerWidth={8}
      />
    );
  }
  if (s.type === 'pin') {
    return (
      <Group key={s.id} x={s.x} y={s.y}>
        <Circle radius={14} fill="#ef4444" />
        <Text
          text={String(s.pinNumber ?? '?')}
          fontSize={12}
          fontStyle="bold"
          fill="white"
          align="center"
          verticalAlign="middle"
          x={-14}
          y={-8}
          width={28}
          height={16}
        />
      </Group>
    );
  }
  return null;
}
```

```tsx
}

const POPUP_W = 280;
const POPUP_H = 150;

function calcPosition(
  shape: DrawShape,
  cw: number,
  ch: number
): { x: number; y: number } {
  let cx = shape.x;
  let cy = shape.y;

  if (shape.type === 'rect') {
    cx = shape.x + (shape.width ?? 0) / 2;
    cy = shape.y + Math.abs(shape.height ?? 0) + 10;
  } else if (shape.type === 'arrow' && shape.points) {
    cx = shape.points[2];
    cy = shape.points[3] + 10;
  } else if (shape.type === 'pin') {
    cx = shape.x;
    cy = shape.y + 24;
  }

  const x = Math.max(8, Math.min(cx - POPUP_W / 2, cw - POPUP_W - 8));
  const y = Math.max(8, Math.min(cy, ch - POPUP_H - 8));
  return { x, y };
}

declare global {
  interface Window {
    SpeechRecognition?: typeof SpeechRecognition;
    webkitSpeechRecognition?: typeof SpeechRecognition;
  }
}

export function ShapeAnnotation({
  shape,
  containerWidth,
  containerHeight,
  onConfirm,
  onDismiss,
}: ShapeAnnotationProps) {
  const [text, setText] = useState('');
  const [listening, setListening] = useState(false);
  const recRef = useRef<SpeechRecognition | null>(null);
  const { x, y } = calcPosition(shape, containerWidth, containerHeight);

  const toggleVoice = () => {
    const SpeechRec =
      window.SpeechRecognition ?? window.webkitSpeechRecognition;
    if (!SpeechRec) return;

    if (listening) {
      recRef.current?.stop();
      setListening(false);
      return;
    }

    const rec = new SpeechRec();
    rec.lang = 'uk-UA';
    rec.interimResults = false;
    rec.onresult = (e) =>
      setText((prev) => (prev + ' ' + e.results[0][0].transcript).trim());
    rec.onend = () => setListening(false);
    rec.start();
    recRef.current = rec;
    setListening(true);
  };

  return (
    <div
      className="absolute z-[10002] bg-white rounded-[16px] shadow-[0_8px_32px_rgba(0,0,0,0.2)] p-3 w-[280px]"
      style={{ left: x, top: y }}
      onClick={(e) => e.stopPropagation()}
    >
      <p className="text-[11px] font-bold text-[#9a9a9a] uppercase tracking-wider mb-2">
        Вкажіть причину...
      </p>
      <textarea
        value={text}
        onChange={(e) => setText(e.target.value)}
        placeholder="Опишіть проблему..."
        rows={3}
        autoFocus
        className="w-full bg-[#f4f4f5] rounded-[10px] text-[13px] text-[#1f1f1f] placeholder:text-[#a3a3a3] resize-none outline-none border border-transparent focus:border-[#1f1f1f] p-[10px] transition-colors"
      />
      <div className="flex gap-2 mt-2">
        <button
          onClick={toggleVoice}
          title="Диктувати голосом"
          className={`w-[36px] h-[36px] rounded-[10px] flex items-center justify-center transition-colors ${
            listening
              ? 'bg-[#ef4444] text-white'
              : 'bg-[#f4f4f5] text-[#9a9a9a] hover:bg-[#e9e9e9]'
          }`}
        >
          {listening ? <MicOff size={14} /> : <Mic size={14} />}
        </button>
        <button
          onClick={onDismiss}
          className="flex-1 h-[36px] rounded-[10px] text-[13px] font-bold bg-[#f5f5f5] text-[#1f1f1f] hover:bg-[#ebebeb] transition-colors"

          Скасувати
```

```tsx
}

export function CaptureMode({ onSave, onCancel }: CaptureModeProps) {
  const [screenshotUrl, setScreenshotUrl] = useState<string | null>(null);
  const [tool, setTool] = useState<DrawTool>('rect');
  const [shapes, setShapes] = useState<DrawShape[]>([]);
  const [annotations, setAnnotations] = useState<Record<string, string>>({});
  const [pendingShape, setPendingShape] = useState<DrawShape | null>(null);
  const w = typeof window !== 'undefined' ? window.innerWidth : 1280;
  const h = typeof window !== 'undefined' ? window.innerHeight : 800;

  useEffect(() => {
    // Small delay so the overlay itself doesn't appear in the screenshot
    const timer = setTimeout(async () => {
      const { default: html2canvas } = await import('html2canvas');
      const canvas = await html2canvas(document.body, {
        useCORS: true,
        allowTaint: true,
        logging: false,
        ignoreElements: (el) => el.getAttribute('data-buggy-bag') === 'true',
      });
      setScreenshotUrl(canvas.toDataURL('image/png'));
    }, 80);
    return () => clearTimeout(timer);
  }, []);

  const handleShapeComplete = useCallback((shape: DrawShape) => {
    setShapes((prev) => [...prev, shape]);
    setPendingShape(shape);
  }, []);

  const handleAnnotationConfirm = useCallback(
    (shapeId: string, text: string) => {
      setAnnotations((prev) => ({ ...prev, [shapeId]: text }));
      setPendingShape(null);
    },
    []
  );

  const handleAnnotationDismiss = useCallback(() => {
    if (pendingShape) {
      setShapes((prev) => prev.filter((s) => s.id !== pendingShape.id));
    }
    setPendingShape(null);
  }, [pendingShape]);

  const handleSave = useCallback(() => {
    if (!screenshotUrl) return;
    onSave({ screenshotDataUrl: screenshotUrl, shapes, annotations });
  }, [screenshotUrl, shapes, annotations, onSave]);

  return (
    <div
      data-buggy-bag="true"
      className="fixed inset-0 z-[10000]"
      style={{ userSelect: 'none' }}
    >
      {/* Screenshot background */}
      {screenshotUrl ? (
        <img
          src={screenshotUrl}
          alt="page screenshot"
          className="absolute inset-0 w-full h-full object-cover pointer-events-none"
          draggable={false}
        />
      ) : (
        <div className="absolute inset-0 bg-black/50 flex items-center justify-center">
          <span className="text-white text-[16px] font-bold animate-pulse">
            Знімок екрану...
          </span>
        </div>
      )}

      {/* Drawing canvas (only after screenshot ready) */}
      {screenshotUrl && !pendingShape && (
        <DrawingCanvas
          width={w}
          height={h}
          tool={tool}
          shapes={shapes}
          onShapeComplete={handleShapeComplete}
        />
      )}

      {/* Annotation popup */}
      {pendingShape && (
        <ShapeAnnotation
          shape={pendingShape}
          containerWidth={w}
          containerHeight={h}
          onConfirm={handleAnnotationConfirm}
          onDismiss={handleAnnotationDismiss}
        />
      )}

      {/* Toolbar */}
      {screenshotUrl && (
        <DrawingToolbar
          activeTool={tool}
          onToolChange={setTool}
          onSave={handleSave}
          onCancel={onCancel}
        />
      )}
    </div>
  );
}
```

```tsx
  const [copied, setCopied] = useState(false);
  const [error, setError] = useState<string | null>(null);

    setLoading(true);
    setError(null);
    try {
      const res = await fetch('/api/generate-ai-prompt', {
        method: 'POST',
        headers: { 'Content-Type': 'application/json' },
        body: JSON.stringify({
          bugs: activeBugs.map((b) => ({
            id: b.id,
            screenshotDataUrl: b.screenshotDataUrl,
            annotations: b.annotations,
            shapes: b.shapes,
            createdAt: b.createdAt,
          })),
        }),
      });
      if (!res.ok) throw new Error(`Server error: ${res.status}`);
      const data = await res.json();
      setPrompt(data.prompt ?? JSON.stringify(data, null, 2));

      setLoading(false);
```
